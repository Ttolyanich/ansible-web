Онбординг нового хоста (Linux)

Подготовка доступа, установка Zabbix Agent 2 (7.0 LTS) и заведение пользователей

1 Выполните команду на новом сервере:

Работает и под root, и через sudo. В чистом Debian сама проверит и установит пакет sudo, создаст пользователя ansible, пропишет SSH-ключ и права. 

U="ansible"
PUBKEY="ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIJW6ABu7Ehczk260wGUp/am9lPeXmqz0TZa2pDulC6PW ITSG-ansible"

if ! command -v sudo >/dev/null 2>&1; then
  if [ "$(id -u)" -ne 0 ]; then
    echo "ОШИБКА: sudo не установлен. Выполните команду под root: su -" >&2
    exit 1
  fi
  echo "Утилита sudo не найдена. Устанавливаем sudo..."
  if command -v apt-get >/dev/null 2>&1; then
    apt-get update -qq && DEBIAN_FRONTEND=noninteractive apt-get install -y -qq sudo
  elif command -v dnf >/dev/null 2>&1; then
    dnf install -y sudo
  elif command -v yum >/dev/null 2>&1; then
    yum install -y sudo
  elif command -v apk >/dev/null 2>&1; then
    apk add sudo
  fi
fi

SUDO=""
[ "$(id -u)" -ne 0 ] && SUDO="sudo"
$SUDO id "$U" >/dev/null 2>&1 || $SUDO useradd -m -s /bin/bash "$U" 2>/dev/null || $SUDO adduser -D -s /bin/bash "$U" 2>/dev/null
$SUDO install -d -m 700 -o "$U" -g "$U" "/home/$U/.ssh"
echo "$PUBKEY" | $SUDO tee "/home/$U/.ssh/authorized_keys" > /dev/null
$SUDO chown -R "$U:$U" "/home/$U/.ssh"
$SUDO chmod 600 "/home/$U/.ssh/authorized_keys"
$SUDO mkdir -p /etc/sudoers.d
$SUDO grep -q "includedir /etc/sudoers.d" /etc/sudoers 2>/dev/null || echo "#includedir /etc/sudoers.d" | $SUDO tee -a /etc/sudoers >/dev/null
echo "$U ALL=(ALL:ALL) NOPASSWD:ALL" | $SUDO tee "/etc/sudoers.d/$U" > /dev/null
$SUDO chmod 0440 "/etc/sudoers.d/$U"
echo "Готово! Пользователь $U успешно настроен."
